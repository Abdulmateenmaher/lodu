    <script>
        const dice1 = document.getElementById('dice1');
        const dice2 = document.getElementById('dice2');
        const historyRow = document.getElementById('history-row');

        let isRolling = false;

        const rotations = {
            1: 'rotateX(0deg) rotateY(0deg)',
            6: 'rotateX(0deg) rotateY(180deg)',
            3: 'rotateX(0deg) rotateY(-90deg)',
            4: 'rotateX(0deg) rotateY(90deg)',
            2: 'rotateX(-90deg) rotateY(0deg)',
            5: 'rotateX(90deg) rotateY(0deg)'
        };

        const dotLayouts = {
            1: [4],
            2: [0, 8],
            3: [0, 4, 8],
            4: [0, 2, 6, 8],
            5: [0, 2, 4, 6, 8],
            6: [0, 2, 3, 5, 6, 8]
        };

        function createMiniDice(num) {
            const dice = document.createElement('div');
            dice.className = 'history-mini-dice';
            // Create 9 grid slots
            for (let i = 0; i < 9; i++) {
                const slot = document.createElement('div');
                if (dotLayouts[num].includes(i)) {
                    const dot = document.createElement('div');
                    dot.className = 'mini-dot';
                    slot.appendChild(dot);
                }
                dice.appendChild(slot);
            }
            return dice;
        }

        function addHistoryResult(res1, res2) {
            const pairContainer = document.createElement('div');
            pairContainer.className = 'history-pair';
            
            // Add double highlight if needed
            if (res1 === res2) {
                pairContainer.style.background = 'rgba(255, 255, 255, 0.15)';
            }

            pairContainer.appendChild(createMiniDice(res1));
            pairContainer.appendChild(createMiniDice(res2));

            if (historyRow.children.length >= 8) {
                historyRow.removeChild(historyRow.firstChild);
            }
            
            historyRow.appendChild(pairContainer);
            historyRow.scrollLeft = historyRow.scrollWidth;
        }

        function rollDice() {
            if (isRolling) return;
            isRolling = true;

            dice1.classList.add('rolling');
            dice2.classList.add('rolling');
            
            const res1 = Math.floor(Math.random() * 6) + 1;
            const res2 = Math.floor(Math.random() * 6) + 1;

            setTimeout(() => {
                dice1.classList.remove('rolling');
                dice2.classList.remove('rolling');
                
                dice1.style.transform = rotations[res1];
                dice2.style.transform = rotations[res2];
                
                setTimeout(() => {
                    addHistoryResult(res1, res2);
                    isRolling = false;
                }, 100);
            }, 300);
        }

        dice1.style.transform = 'rotateX(-20deg) rotateY(20deg)';
        dice2.style.transform = 'rotateX(-20deg) rotateY(-20deg)';
    </script>
